13	def extract_json_field(text: str, field: str, default: str = "") -> str:
14	    """Helper function to extract a field from JSON in LLM response."""
15	    # Try direct parse
16	    try:
17	        data = json.loads(text)
18	        if isinstance(data, dict):
19	            return str(data.get(field, default))
20	    except json.JSONDecodeError:
21	        pass
22	
23	    # Try code blocks: ```json ... ``` or ``` ... ```
24	    for match in re.finditer(r"```(?:json)?\s*([\s\S]*?)\s*```", text):
25	        try:
26	            data = json.loads(match.group(1))
27	            if isinstance(data, dict):
28	                return str(data.get(field, default))
29	        except json.JSONDecodeError:
30	            pass
31	
32	    # Find balanced braces and try parsing
33	    for start in range(len(text)):
34	        if text[start] != "{":
35	            continue
36	        depth, pos, in_str = 1, start + 1, False
37	        while pos < len(text) and depth > 0:
38	            c = text[pos]
39	            if c == '"' and (pos == 0 or text[pos - 1] != "\\"):
40	                in_str = not in_str
41	            elif not in_str:
42	                depth += 1 if c == "{" else (-1 if c == "}" else 0)
43	            pos += 1
44	        if depth == 0:
45	            candidate = text[start:pos]
46	            candidate = re.sub(
47	                r",\s*([\]}])", r"\1", candidate
48	            )  # Remove trailing commas
49	            try:
50	                data = json.loads(candidate)
51	                if isinstance(data, dict):
52	                    return str(data.get(field, default))
53	            except json.JSONDecodeError:
54	                pass
55	
56	    # Regex fallback
57	    match = re.findall(rf'"{field}"\s*:\s*"([^"]*)"', text)
58	    return match[-1] if match else default
