"""Prototype: contrastive error-driven memory."""
import random

random.seed(42)
examples = [
    {"input": f"Classify: text_{i}", "target": f"Class{i%5}", 
     "prediction": f"Class{(i+1)%5}" if i % 3 == 0 else f"Class{i%5}",
     "was_correct": i % 3 != 0}
    for i in range(30)
]

MAX_CHARS = 30000

def format_as_contrastive(errors, successes, max_chars=MAX_CHARS):
    """Mix contrastive pairs (errors) with plain Q/A (successes)."""
    parts = []
    total = 0
    # Show recent errors first with contrastive framing
    for ex in reversed(errors[-15:]):
        part = (f"Q: {ex['input']}\n"
                f"  ✗ Wrong: {ex['prediction']}\n"
                f"  ✓ Correct: {ex['target']}")
        if total + len(part) > max_chars: break
        parts.append(part)
        total += len(part) + 2
    # Then recent successes (plain)
    for ex in reversed(successes[-8:]):
        part = f"Q: {ex['input']}\nA: {ex['target']}"
        if total + len(part) > max_chars: break
        parts.append(part)
        total += len(part) + 2
    return "\n\n".join(parts)

errors = [e for e in examples if not e["was_correct"]]
successes = [e for e in examples if e["was_correct"]]

section = format_as_contrastive(errors, successes)
print(f"Errors: {len(errors)}, Successes: {len(successes)}")
print(f"Section length: {len(section)} chars")
print("---SAMPLE---")
print(section[:600])
print("OK - contrastive format verified")
EOF
