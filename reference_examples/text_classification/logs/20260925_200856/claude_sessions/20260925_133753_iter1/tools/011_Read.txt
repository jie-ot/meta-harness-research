66	def parse_harmony_response(raw_content: str) -> str:
67	    """Parse GPT-OSS harmony format, extracting final channel content.
68	
69	    Returns the 'final' channel content, or raw content if parsing fails.
70	    """
71	    enc = _get_harmony_enc()
72	    try:
73	        tokens = enc.encode(raw_content, allowed_special="all")
74	        parsed = enc.parse_messages_from_completion_tokens(
75	            tokens, role=Role.ASSISTANT, strict=False
76	        )
77	
78	        for msg in parsed:
79	            if msg.channel == "final":
80	                return "".join(c.text for c in msg.content if hasattr(c, "text"))
81	
82	        if parsed:
83	            return "".join(c.text for c in parsed[-1].content if hasattr(c, "text"))
84	    except Exception:
85	        pass

112	def _completion_cost_usd(response: Any, usage: Any, allow_price_table: bool) -> float:
113	    """Prefer the provider's own cost, then LiteLLM's price table."""
114	    reported = []
115	    if usage is not None:
116	        reported.append(getattr(usage, "cost", None))
117	        extra = getattr(usage, "model_extra", None) or {}
118	        if isinstance(extra, dict):
119	            reported.append(extra.get("cost"))
120	    hidden = getattr(response, "_hidden_params", None) or {}
121	    if isinstance(hidden, dict):
122	        reported.append(hidden.get("response_cost"))
123	    for value in reported:
124	        if value is None:
125	            continue
126	        try:
127	            return float(value)
128	        except (TypeError, ValueError):
129	            continue
130	    if not allow_price_table:
131	        return 0.0
132	    try:
133	        return float(completion_cost(completion_response=response) or 0.0)
134	    except Exception:
135	        return 0.0

138	def _extract_content(response: Any) -> str:
139	    message = response.choices[0].message
140	    content = getattr(message, "content", None)
141	    if isinstance(content, str):
142	        return content
143	    if isinstance(content, list):
144	        parts = []
145	        for item in content:
146	            if isinstance(item, dict) and "text" in item:
147	                parts.append(item["text"])
148	            elif hasattr(item, "text"):
149	                parts.append(item.text)
150	        return "".join(parts)
151	    return ""
