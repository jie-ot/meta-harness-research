"""Prototype v2: missed-component boost with component-level matching."""
import re, math
from collections import defaultdict

DELIM_RE = re.compile(r'[;；]')
WRAPPER_RE = re.compile(r'^\[[^\]]*\]|<eoa>\s*$|\[[^\]]*\]\s*$', re.IGNORECASE)

def clean(label):
    return WRAPPER_RE.sub('', label).strip()

def split_label(label):
    c = clean(label)
    parts = [p.strip() for p in DELIM_RE.split(c) if p.strip()]
    return parts if parts else [c]

def compute_missed(pred, gt):
    pred_parts = set(split_label(pred))
    gt_parts = set(split_label(gt))
    return list(gt_parts - pred_parts)

def example_missed_overlap(example_target, missed_counts):
    """Sum of missed_counts for components found in this example's target."""
    parts = set(split_label(example_target))
    return sum(missed_counts.get(p, 0) for p in parts)

# Real training errors from LawBench
errors = [
    ("[罪名]诈骗<eoa>",                          "诈骗;组织、领导传销活动;销售假冒注册商标的商品"),
    ("[罪名]故意伤害;毁坏财物<eoa>",                "故意伤害;非法持有枪支"),
    ("[罪名]强迫卖淫<eoa>",                        "组织卖淫;强迫卖淫;强奸"),
    ("[罪名]走私<eoa>",                            "走私贵重金属;走私普通货物、物品"),
    ("[罪名]诈骗<eoa>",                            "信用卡诈骗"),
    ("[罪名]故意伤害<eoa>",                         "故意伤害"),  # correct
    ("[罪名]侵犯著作权<eoa>",                       "侵犯著作权;非法经营"),
]

missed_counts = defaultdict(float)
DECAY = 0.95
for pred, gt in errors:
    missed = compute_missed(pred, gt)
    for m in missed:
        missed_counts[m] += 1.0

print("Missed components:", dict(sorted(missed_counts.items(), key=lambda x: -x[1])))

# Simulated example pool with realistic targets
pool = [
    {"target": "组织卖淫;强迫卖淫", "sim": 0.4},
    {"target": "非法经营",           "sim": 0.3},
    {"target": "诈骗;信用卡诈骗",    "sim": 0.6},
    {"target": "故意伤害",           "sim": 0.7},
    {"target": "非法持有枪支",       "sim": 0.25},
    {"target": "走私贵重金属",       "sim": 0.2},
    {"target": "销售假冒注册商标的商品", "sim": 0.15},
]

ALPHA = 0.3
print("\nScores (sim + alpha * log(1 + overlap)):")
for ex in pool:
    overlap = example_missed_overlap(ex["target"], missed_counts)
    boosted = ex["sim"] + ALPHA * math.log(1 + overlap)
    print(f"  {ex['target']:30s}  sim={ex['sim']:.2f}  overlap={overlap:.1f}  final={boosted:.3f}")

# Rank by final score
ranked = sorted(pool, key=lambda ex: ex["sim"] + ALPHA * math.log(1 + example_missed_overlap(ex["target"], missed_counts)), reverse=True)
print("\nRanked:")
for i, ex in enumerate(ranked):
    print(f"  {i+1}. {ex['target']}")
