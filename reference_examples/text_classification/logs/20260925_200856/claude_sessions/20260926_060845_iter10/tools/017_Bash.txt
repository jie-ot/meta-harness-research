# Test the ordering logic: fill similarity first, append confusion last

examples = [
    {'input': f'q{i}', 'target': f'label_{i%4}', 'tokens': frozenset(f'q{i} word'.split())} 
    for i in range(20)
]

import re
from collections import defaultdict

def jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

MAX_CHARS = 500  # small for test

def build_parts_old(query, examples, confusion):
    '''Confusion first, then fill (current approach)'''
    q_tok = frozenset(re.findall(r'[A-Za-z0-9]+', query.lower()))
    ranked = sorted(enumerate(examples), key=lambda x: jaccard(q_tok, x[1]['tokens']), reverse=True)
    
    # get confusion targets for top-3 candidate labels

            if actual not in top_labels:

    parts = []
    used = set()
    total = 0
    
    # inject confusion examples FIRST
    if confused_with:
        per_label = defaultdict(list)
        for idx, ex in ranked:
            if ex['target'] in confused_with:
                per_label[ex['target']].append((idx, ex))

        confusion_reserve = sum(len(f'Q: {ex[\"input\"]}\nA: {ex[\"target\"]}') + 2 for _, ex in sampled)
    
    fill_budget = MAX_CHARS - confusion_reserve
