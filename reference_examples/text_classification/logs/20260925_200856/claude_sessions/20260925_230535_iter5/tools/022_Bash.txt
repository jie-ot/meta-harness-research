def tokenize(text):
    return frozenset(re.findall(r'[A-Za-z0-9]+', text.lower()))

def jaccard(a, b):
    if not a and not b:
        return 0.0
    return len(a & b) / len(a | b)

def mmr_fill(query_toks, candidates, already_used, budget=5000, lam=0.6):
    selected_examples = []
    remaining = [(s, i, e) for s, i, e in candidates if i not in already_used]
    total_chars = 0
    while remaining and total_chars < budget:
        best_score, best_item = -999, None
        for s, i, ex in remaining:
            part = f'Q: {ex[\"raw_question\"]}\nA: {ex[\"target\"]}'
            if total_chars + len(part) + 2 > budget:

                (jaccard(ex['tokens'], se['tokens']) for se in selected_examples),
                default=0.0
