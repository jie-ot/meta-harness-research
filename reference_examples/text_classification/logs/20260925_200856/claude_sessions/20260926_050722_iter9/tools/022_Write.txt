"""Prototype: recency-weighted Jaccard retrieval. Compare 3 variants."""
import re
from collections import defaultdict

# Simulate 20 stored examples (ascending index = more recent)
# Construct a scenario where recent examples are more relevant
import random
random.seed(42)

def make_tokens(text):
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def jaccard(a, b):
    if not a and not b:
        return 0.0
    return len(a & b) / len(a | b)

# Simulate examples with known relevance (ground truth similarity to query)
# Older examples (indices 0-9) are from distribution A (low overlap with query)
# Newer examples (indices 10-19) are from distribution B (high overlap with query)
# We expect recency weighting to promote B examples

vocab_old = ["alpha", "beta", "gamma", "delta", "epsilon", "zeta", "eta", "theta"]
vocab_new = ["apple", "banana", "cherry", "date", "elderberry", "fig", "grape", "honey"]
vocab_query = ["apple", "cherry", "fig", "grape", "mango", "kiwi"]  # overlaps with vocab_new

examples = []
for i in range(10):  # old examples, low relevance to query
    words = random.sample(vocab_old, 4)
    examples.append({"idx": i, "tokens": make_tokens(" ".join(words)), "label": f"old_{i%3}"})

for i in range(10, 20):  # new examples, high relevance to query
    words = random.sample(vocab_new, 4)
    examples.append({"idx": i, "tokens": make_tokens(" ".join(words)), "label": f"new_{i%3}"})

query_tokens = make_tokens(" ".join(vocab_query))
N = len(examples)

# Variant 1: pure Jaccard (baseline - current behavior)
def variant_pure_jaccard(examples, query_tokens, k=5):
    scored = [(jaccard(query_tokens, ex["tokens"]), ex["idx"], ex) for ex in examples]
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return scored[:k]

# Variant 2: linear recency × Jaccard
# weight = (idx+1)/N scales 0.05→1.0; blend: sim * (0.3 + 0.7 * recency)
def variant_recency_blend(examples, query_tokens, k=5, alpha=0.5):
    N = len(examples)
    scored = []
    for ex in examples:
        sim = jaccard(query_tokens, ex["tokens"])
        recency = (ex["idx"] + 1) / N
        combined = sim * (1 - alpha + alpha * recency)
        scored.append((combined, ex["idx"], ex))
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return scored[:k]

# Variant 3: separate rank combination (Reciprocal Rank Fusion)
def variant_rrf(examples, query_tokens, k=5, c=60):
    N = len(examples)
    # Rank by similarity
    sim_ranked = sorted(examples, key=lambda ex: jaccard(query_tokens, ex["tokens"]), reverse=True)
    sim_rank = {ex["idx"]: r+1 for r, ex in enumerate(sim_ranked)}
    # Rank by recency (higher idx = more recent = better)
    rec_ranked = sorted(examples, key=lambda ex: ex["idx"], reverse=True)
    rec_rank = {ex["idx"]: r+1 for r, ex in enumerate(rec_ranked)}

    # RRF score
    scored = []
    for ex in examples:
        rrf = 1.0/(c + sim_rank[ex["idx"]]) + 1.0/(c + rec_rank[ex["idx"]])
        scored.append((rrf, ex["idx"], ex))
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return scored[:k]

def evaluate(top_k):
    # Count how many of top-k are "new" (high-relevance) examples
    new_count = sum(1 for _, idx, _ in top_k if idx >= 10)
    sim_vals = [jaccard(query_tokens, ex["tokens"]) for _, _, ex in top_k]
    avg_sim = sum(sim_vals) / len(sim_vals)
    labels = [ex["label"] for _, idx, ex in top_k]
    return new_count, avg_sim, labels

print("Query vocab:", vocab_query)
print(f"N examples: {N} (0-9 old/low-relevance, 10-19 new/high-relevance)")
print()

# Try multiple alpha values for recency blend
for alpha in [0.2, 0.4, 0.6]:
    top_k = variant_recency_blend(examples, query_tokens, k=5, alpha=alpha)
    new_count, avg_sim, labels = evaluate(top_k)
    print(f"Blend alpha={alpha}: new_examples={new_count}/5, avg_sim={avg_sim:.3f}, labels={labels}")

print()
top_k_base = variant_pure_jaccard(examples, query_tokens, k=5)
new_count, avg_sim, labels = evaluate(top_k_base)
print(f"Pure Jaccard: new_examples={new_count}/5, avg_sim={avg_sim:.3f}, labels={labels}")

print()
top_k_rrf = variant_rrf(examples, query_tokens, k=5)
new_count, avg_sim, labels = evaluate(top_k_rrf)
print(f"RRF: new_examples={new_count}/5, avg_sim={avg_sim:.3f}, labels={labels}")

print()
# Now test a scenario where old examples ARE more relevant (reverse test)
# This ensures recency weighting doesn't destroy performance when old examples are better
print("=== Reverse test: old examples ARE more relevant ===")
query_tokens_old = make_tokens(" ".join(random.sample(vocab_old, 4)))
print(f"Query tokens: {query_tokens_old}")

for alpha in [0.2, 0.4, 0.6]:
    top_k = variant_recency_blend(examples, query_tokens_old, k=5, alpha=alpha)
    new_count, avg_sim, labels = evaluate(top_k)
    old_count = 5 - new_count
    print(f"Blend alpha={alpha}: old_examples(relevant)={old_count}/5, avg_sim={avg_sim:.3f}")

top_k_base = variant_pure_jaccard(examples, query_tokens_old, k=5)
new_count, avg_sim, labels = evaluate(top_k_base)
print(f"Pure Jaccard: old_examples(relevant)={5-new_count}/5, avg_sim={avg_sim:.3f}")
