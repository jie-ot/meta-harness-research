def count_render(pool, budget, sketch=None):
    total, count = 0, 0
    for e in pool:
        raw_q = e.get('raw_question', e.get('input', ''))
        if sketch == 'A':
            raw_q = raw_q[:SKETCH_CHARS_A]
        elif sketch == 'B' and len(raw_q) > LONG_THRESHOLD:
            raw_q = raw_q[:SKETCH_CHARS_A]
        part = f"Q: {raw_q}\nA: {e['target']}"
        if total + len(part) + 2 > budget:
            break
        total += len(part) + 2
        count += 1
    return count
