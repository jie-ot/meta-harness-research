def _tok_word(t): return frozenset(re.findall(r'[A-Za-z0-9]+', t.lower()))
def _tok_bigram(t):
    t2 = re.sub(r'\s+', ' ', t.strip())
    return frozenset(t2[i:i+2] for i in range(len(t2)-1)) if len(t2)>=2 else frozenset()

def _tok_notation_adaptive(t):

    total = len(t.replace(' ',''))
