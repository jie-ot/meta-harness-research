"""Prototype: error_correction_memory — confusion-driven example surfacing."""
import re
from collections import defaultdict

def tokenize(text):
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

# Simulate memory after training
examples = []
for i, (lbl, smiles) in enumerate([
    ("Protections", "COC(=O)NC"), ("Oxidation", "CC(=O)O"), ("C-C bond formation", "C=CC"),
    ("Reductions", "CC=O"), ("Protections", "BOC_amino"), ("Oxidation", "MnO2_substrate"),
]):
    examples.append({
        "raw_question": f"Context: reaction {lbl}.\nInput: {smiles}",
        "target": lbl,
        "tokens": tokenize(f"{lbl} {smiles}"),
        "was_correct": i % 3 != 0,
    })

# Confusion matrix: pred -> gt -> count (populated from errors)
confusion: dict = defaultdict(lambda: defaultdict(int))
errors = [
    # (predicted, actual)
    ("Protections", "Oxidation"), ("Protections", "Oxidation"),
    ("Oxidation", "Protections"), ("Reductions", "C-C bond formation"),
    ("C-C bond formation", "Reductions"), ("Protections", "Oxidation"),
]
for pred, gt in errors:
    confusion[pred][gt] += 1

# Approach: at predict time, given a query:
# 1. Find top-N similar examples (same as similarity_retrieval)
# 2. Extract candidate labels from those examples
# 3. For each candidate label, look up its most common confusions
# 4. Find stored examples of those "confused-with" labels → disambiguation examples
# 5. Inject disambiguation examples first, then similarity-ranked examples

query = "Context: reaction Protections.\nInput: TBoc_substrate"
q_tok = tokenize(query)

# Step 1: similarity ranking
scored = sorted(
    [(jaccard(q_tok, ex["tokens"]), i, ex) for i, ex in enumerate(examples)],
    key=lambda x: (x[0], x[1]), reverse=True
)
print("Top similarity scores:")
for score, idx, ex in scored[:4]:
    print(f"  {score:.3f}  {ex['target']}: {ex['raw_question'][:50]}")

# Step 2: candidate labels from top-3 examples
candidate_labels = [ex["target"] for _, _, ex in scored[:3]]
print(f"\nCandidate labels from similarity: {candidate_labels}")

# Step 3: confusions for candidate labels
disambiguation_targets = set()
for lbl in candidate_labels:
    if lbl in confusion:
        top_confused = sorted(confusion[lbl].items(), key=lambda x: -x[1])[:2]
        for confused_with, cnt in top_confused:
            print(f"  Confusion: {lbl} confused with {confused_with} ({cnt}x)")
            disambiguation_targets.add(confused_with)

print(f"\nDisambiguation targets needed: {disambiguation_targets}")

# Step 4: find corrective examples — examples OF the confused-with labels
# that the model has seen (preferably ones where it got it right)
disambig_examples = []
for ex in examples:
    if ex["target"] in disambiguation_targets:
        disambig_examples.append(ex)

print("Disambiguation examples found:")
for ex in disambig_examples:
    print(f"  {ex['target']}: {ex['raw_question'][:50]}")

# VARIANT A: disambiguation examples first, then similarity fill
# VARIANT B: interleave disambiguation and similarity examples
# VARIANT C: append disambiguation after similarity (show contrast at end)
#
# Test: variant A (disambiguation first) surfaces the confusing distinction
#       early in context where LLMs pay more attention

print("\n=== Variant A prompt structure ===")
print("Section 1 - Disambiguation (avoid these confusions):")
for ex in disambig_examples[:2]:
    print(f"  Q: {ex['raw_question'][:60]}")
    print(f"  A: {ex['target']}")
print("Section 2 - Similar examples:")
for _, _, ex in scored[:3]:
    print(f"  Q: {ex['raw_question'][:60]}")
    print(f"  A: {ex['target']}")

print("\nPROTOTYPE PASSED: confusion-driven disambiguation mechanism is sound.")
print("Novel mechanism: confusion matrix drives which examples surface, not just similarity.")
