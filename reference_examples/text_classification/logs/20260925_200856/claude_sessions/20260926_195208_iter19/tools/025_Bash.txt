"""Prototype: context_quality_confusion_memory
Test that confusion weights vary by retrieval quality and
produce different disambiguation priorities vs uniform weights.
"""
import sys, json, re, os
from collections import defaultdict

base = r'D:\我的文件\华中科技大学\李老师课题组研究工作\meta-harness\reference_examples\text_classification\logs\20260925_200856'

_CJK_RE = re.compile(r'[一-鿿぀-ゟ゠-ヿ＀-￯]')
_WRAPPER_RE = re.compile(r'^\s*\[[^\]]*\]|<eoa>\s*$|\[[^\]]*\]\s*$', re.IGNORECASE)

def has_cjk(t): return bool(_CJK_RE.search(t))
def tok_word(t): return frozenset(re.findall(r"[A-Za-z0-9]+", t.lower()))
def tok_bigram(t):
    t = re.sub(r'\s+', ' ', t.strip())
    return frozenset(t[i:i+2] for i in range(len(t)-1)) if len(t)>=2 else frozenset()
def tok(t): return tok_bigram(t) if has_cjk(t) else tok_word(t)
def jaccard(a, b): return len(a&b)/len(a|b) if a or b else 0.0
def normalize(p):
    if not p: return p
    c = _WRAPPER_RE.sub('', p).strip()
    return c if c else p

# Load LawBench log
log_path = os.path.join(base, 'LawBench', 'decayed_confusion_memory', 'gpt-oss-120b', 'log.jsonl')
steps = []
with open(log_path, encoding='utf-8') as f:
    for line in f:
        d = json.loads(line)
        if d.get('type') == 'step':
            steps.append(d)

print(f"Total steps: {len(steps)}")

# Simulate: replay first 100 steps, build two confusion matrices:
# (A) uniform weight 1.0 per error  
# (B) context-quality weighted

# We need to simulate retrieval quality. For each step, we'll approximate
# the similarity by looking at how much the prompt grew vs baseline.
# Actually, we can use prompt_len as a proxy for context quality:
# - prompt_len near baseline (~300) means cold start (no examples yet)
# - prompt_len much larger means rich context

# Compute baseline prompt_len from first few steps
early_lens = [s['prompt_len'] for s in steps[:5]]
BASE_LEN = min(early_lens)
print(f"Baseline prompt_len (cold start): {BASE_LEN}")

conf_uniform = defaultdict(lambda: defaultdict(float))
conf_quality = defaultdict(lambda: defaultdict(float))

for i, step in enumerate(steps[:150]):
    if not step.get('ok', True):
        pred = normalize(step['pred'])
        gt = step['tgt']
        if pred and pred != gt:
            # Uniform weight
            conf_uniform[pred][gt] += 1.0
            
            # Quality weight: based on prompt_len relative to max seen so far
            max_len = max(s['prompt_len'] for s in steps[:i+1])
            # Normalized quality: 0 = cold start, 1 = full context
            quality = (step['prompt_len'] - BASE_LEN) / max(max_len - BASE_LEN, 1)
            quality = max(0.0, min(1.0, quality))
            # Weight: 0.3 for low quality, 2.0 for high quality
            weight = 0.3 + 1.7 * quality
            conf_quality[pred][gt] += weight

# Compare top pairs
def top_pairs(conf, n=8):
    pairs = [(cnt, p, a) for p, acts in conf.items() for a, cnt in acts.items()]
    pairs.sort(reverse=True)
    return pairs[:n]

print("\nTop confusion pairs — UNIFORM:")
for cnt, p, a in top_pairs(conf_uniform):
    print(f"  {cnt:.2f}  {p[:30]} -> {a[:30]}")

print("\nTop confusion pairs — QUALITY-WEIGHTED:")
for cnt, p, a in top_pairs(conf_quality):
    print(f"  {cnt:.2f}  {p[:30]} -> {a[:30]}")

# Count how often top pairs differ
unif_set = {(p,a) for _,p,a in top_pairs(conf_uniform, 5)}
qual_set = {(p,a) for _,p,a in top_pairs(conf_quality, 5)}
print(f"\nTop-5 pairs in common: {len(unif_set & qual_set)}")
print(f"Different between methods: uniform-only={unif_set - qual_set}, qual-only={qual_set - unif_set}")
