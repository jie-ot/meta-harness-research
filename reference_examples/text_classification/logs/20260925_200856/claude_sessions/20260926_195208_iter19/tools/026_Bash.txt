# (how often the query's top sim label appears in confusion lookup result)

DECAY = 0.95
PRUNE = 0.05

def simulate(steps, symmetric=False):
    conf = defaultdict(lambda: defaultdict(float))
    errors = 0
    disambig_fires = 0  # times confusion lookup returns any result
    
    for i, step in enumerate(steps):
        pred = normalize(step['pred'])
        gt = step['tgt']
        ok = step.get('ok', True)
        
        if not ok and pred and pred != gt:
            errors += 1
            conf[pred][gt] += 1.0
            if symmetric:
                conf[gt][pred] += 0.5  # reverse direction at half weight
        
        # Decay after each step (simulating per-batch decay)
        if i % 5 == 4:  # every 5 steps, decay
            for p in list(conf):
                for a in list(conf[p]):
                    conf[p][a] *= DECAY
                    if conf[p][a] < PRUNE:
                        del conf[p][a]
                if not conf[p]:
                    del conf[p]
        
        # Would disambiguation fire for this step?
        # Pretend top candidate label = gt (best case)
        top_label = gt
        if top_label in conf and conf[top_label]:
            disambig_fires += 1
    
    return errors, disambig_fires, len(conf)
