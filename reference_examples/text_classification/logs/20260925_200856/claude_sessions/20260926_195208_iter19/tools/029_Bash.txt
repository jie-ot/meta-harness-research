def run_sim(steps, K):

                scored = sorted([(jaccard(q_tok, t), lbl) for t, lbl in all_ex], reverse=True)
                top5_labels = [lbl for _, lbl in scored[:5]]
                if gt in top5_labels:

                tested += 1
