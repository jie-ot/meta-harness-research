# Prototype temporal split: recent_k=20, prior=all older
# Measure: does interleaving produce different top-K selection than plain?
# Compare: (A) single pool jaccard, (B) split pool interleave

RECENT_K = 20
MAX_SELECTED = 8

def top_examples_single(pool, query_tok, n):
    scored = sorted([(jaccard(query_tok, e['tok']), i) for i, e in enumerate(pool)], reverse=True)
    return [pool[i] for _, i in scored[:n]]

def top_examples_split(pool, query_tok, n):
    if len(pool) <= RECENT_K:
        return top_examples_single(pool, query_tok, n)
    recent = pool[-RECENT_K:]
    prior = pool[:-RECENT_K]
    # Take ceil(n/2) from each, then interleave recent-first
    half = max(1, n // 2)
    from_recent = top_examples_single(recent, query_tok, half)
    from_prior = top_examples_single(prior, query_tok, n - half)
    # Interleave: recent first, prior second (prior is farther from question)
    return from_prior + from_recent  # prior first, recent last (recency bias: recent closest to question)
