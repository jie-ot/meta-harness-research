"""
Prototype: label-name-similarity graph retrieval.
Build a graph where labels share edges if one name contains the other (substring)
or they share significant token overlap (>0.5 Jaccard on CJK bigrams or words).
At predict time: get top similar examples, expand their labels via the graph,
inject examples of graph-neighbor labels before filling with similarity.
Works from step 1 (no error history needed). Targets over-generalization errors.
"""
import json, re
from collections import defaultdict

_CJK_RE = re.compile(r'[一-鿿぀-ゟ゠-ヿ＀-￯]')
_WRAPPER_RE = re.compile(r'^\s*\[[^\]]*\]|<eoa>\s*$|\[[^\]]*\]\s*$', re.IGNORECASE)

def _has_cjk(text):
    return bool(_CJK_RE.search(text))

def _tokenize_adaptive(text):
    if _has_cjk(text):
        t = re.sub(r'\s+', ' ', text.strip())
        return frozenset(t[i:i+2] for i in range(len(t)-1)) if len(t) >= 2 else frozenset()
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def _jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

def _normalize(pred):
    if not pred: return pred
    cleaned = _WRAPPER_RE.sub('', pred).strip()
    return cleaned if cleaned else pred

def _label_name_neighbors(label, all_labels, threshold=0.4):
    """Find labels whose name overlaps significantly with label.
    Two types of edges:
    1. Substring: one contains the other
    2. Token Jaccard >= threshold
    """
    neighbors = []
    l_tok = _tokenize_adaptive(label)
    for other in all_labels:
        if other == label:
            continue
        o_tok = _tokenize_adaptive(other)
        # Substring check
        if label in other or other in label:
            neighbors.append(other)
        elif l_tok and o_tok and _jaccard(l_tok, o_tok) >= threshold:
            neighbors.append(other)
    return neighbors

    examples_raw.append(restored)

# Build label graph
all_labels = list(set(ex['target'] for ex in examples_raw))
print(f"Total unique labels: {len(all_labels)}")

# Show the graph neighbors for some known confusors
for lbl in ['诈骗', '受贿', '故意伤害', '盗窃', '行贿']:
    neighbors = _label_name_neighbors(lbl, all_labels)
    print(f"  {repr(lbl)} neighbors ({len(neighbors)}): {[repr(n) for n in neighbors[:8]]}")

# Build graph
label_graph = {}
for lbl in all_labels:
    label_graph[lbl] = _label_name_neighbors(lbl, all_labels)

print(f"\nLabels with at least 1 neighbor: {sum(1 for n in label_graph.values() if n)}/{len(all_labels)}")
print(f"Labels with 0 neighbors: {sum(1 for n in label_graph.values() if not n)}")

# Simulate: for error cases where pred is substring of gt,

                    if _normalize(s.get('pred','')) in s.get('tgt','') and _normalize(s.get('pred',''))]

print(f"\nSubstring-relationship errors: {len(substring_errors)}")

graph_helped = 0
for s, pred, gt in substring_errors:
    # Look up graph neighbors of pred
    neighbors = label_graph.get(pred, [])
    if gt in neighbors:

# More importantly: at predict time, do we know pred? No.
# But we know top-K retrieved example labels. Expand those via graph.
# Test: does graph expansion of top-3 retrieved labels surface gt more than plain top-8?
def get_top_k_with_graph(query, pool, label_graph_local, k=8, graph_slots=3):
    q_tok = _tokenize_adaptive(query)
    scored = sorted([(_jaccard(q_tok, ex['tokens']), i, ex) for i, ex in enumerate(pool)],
                    key=lambda x: (-x[0], x[1]))
    
    top3_labels = list(dict.fromkeys(ex['target'] for _, _, ex in scored[:5]))[:3]
    
    # Expand via graph
    neighbor_labels = set()
    for lbl in top3_labels:
        for nbr in label_graph_local.get(lbl, []):
            neighbor_labels.add(nbr)
    neighbor_labels -= set(top3_labels)
    
    selected = []
    used_labels = set()
    
    # First: graph neighbors
    for lbl in list(neighbor_labels)[:graph_slots]:
        for _, _, ex in scored:
            if ex['target'] == lbl and lbl not in used_labels:
                selected.append(ex)
                used_labels.add(lbl)
                break
    
    # Then fill with similarity
    for _, _, ex in scored:
        if len(selected) >= k:
            break
        if ex['target'] not in used_labels:
            selected.append(ex)
            used_labels.add(ex['target'])
    
    return selected

print(f"  Graph-expanded: {err_graph}/{len(errors)} = {err_graph/len(errors)*100:.1f}%")
