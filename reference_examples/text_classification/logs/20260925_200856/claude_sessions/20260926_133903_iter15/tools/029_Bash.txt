"""
Prototype: reaction-type cluster retrieval for USPTO.
Extract the reaction category tag from any position in the input,
group stored examples by tag, retrieve within the same tag first.
"""
import json, re
from collections import defaultdict

def _extract_category_tag(text):
    """Extract a short category tag from anywhere in the input text.
    Tries common patterns: 'the reaction type is X', 'category: X', etc.
    Returns lowercased tag or empty string if not found.
    """
    # Pattern 1: 'reaction type is <value>'
    m = re.search(r'reaction type is ([^.\n<]+)', text, re.IGNORECASE)
    if m:
        return m.group(1).strip().lower()
    # Pattern 2: 'category: <value>' or 'type: <value>' on its own line
    m = re.search(r'(?:^|\n)(?:category|type|class|task):\s*([^\n]+)', text, re.IGNORECASE)
    if m:
        return m.group(1).strip().lower()
    # No tag found
    return ''

def _has_cjk(text):
    return bool(re.search(r'[一-鿿぀-ゟ゠-ヿ＀-￯]', text))

def _tokenize_adaptive(text):
    if _has_cjk(text):
        t = re.sub(r'\s+', ' ', text.strip())
        return frozenset(t[i:i+2] for i in range(len(t)-1)) if len(t) >= 2 else frozenset()
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def _jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

def get_top_k_clustered_v2(query_full, pool, k=5, min_cluster_size=1):
    tag = _extract_category_tag(query_full)
    q_tok = _tokenize_adaptive(query_full)
    
    if tag:
        primary = [ex for ex in pool if ex['category_tag'] == tag]
        other = [ex for ex in pool if ex['category_tag'] != tag]
    else:
        primary = pool
        other = []
    
    def score(ex):
        return _jaccard(q_tok, ex['tokens'])
    
    selected = []
    used = set()
    
    # Fill from primary pool first (same category)
    primary_scored = sorted(primary, key=lambda ex: -score(ex))
    for ex in primary_scored:
        if len(selected) >= k: break
        idx = id(ex)
        if idx not in used:
            selected.append(ex)
            used.add(idx)
    
    # Fill remainder from other categories by similarity
    other_scored = sorted(other, key=lambda ex: -score(ex))
    for ex in other_scored:
        if len(selected) >= k: break
        idx = id(ex)
        if idx not in used:
            selected.append(ex)
            used.add(idx)
    
    return selected

def get_top_k_flat(query, pool, k=5):
    q_tok = _tokenize_adaptive(query)
    scored = sorted(pool, key=lambda ex: -_jaccard(q_tok, ex['tokens']))
    return scored[:k]
