"""
Prototype: input-segmented dual similarity.
Key idea: split input into prefix (domain header) and suffix (key entity/SMILES/specific fact).
Score examples as max(sim_prefix, sim_suffix).
"""
import json, re
from collections import defaultdict

_CJK_RE = re.compile(r'[一-鿿぀-ゟ゠-ヿ＀-￯]')

def _has_cjk(text):
    return bool(_CJK_RE.search(text))

def _tokenize_word(text):
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def _tokenize_bigram(text):
    t = re.sub(r'\s+', ' ', text.strip())
    if len(t) < 2:
        return frozenset([t]) if t else frozenset()
    return frozenset(t[i:i+2] for i in range(len(t)-1))

def _tokenize_adaptive(text):
    return _tokenize_bigram(text) if _has_cjk(text) else _tokenize_word(text)

def _jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

def _split_input(text, suffix_chars=500):
    """Split into prefix and suffix. Suffix captures the most specific part."""
    if len(text) <= suffix_chars:
        return text, text
    return text[:-suffix_chars], text[-suffix_chars:]

# Measure: does segmented similarity improve top-1 hit rate vs flat similarity?
def get_top_k_flat(query, examples, k=5):
    q_tok = _tokenize_adaptive(query)
    scored = [(_jaccard(q_tok, ex['tokens']), i, ex) for i, ex in enumerate(examples)]
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return [ex for _, _, ex in scored[:k]]

def get_top_k_segmented(query, examples, k=5):
    prefix, suffix = _split_input(query)
    q_prefix = _tokenize_adaptive(prefix)
    q_suffix = _tokenize_bigram(suffix)  # always bigram for suffix
    scored = []
    for i, ex in enumerate(examples):
        sim_prefix = _jaccard(q_prefix, ex['tokens_prefix'])
        sim_suffix = _jaccard(q_suffix, ex['tokens_suffix'])
        combined = max(sim_prefix, sim_suffix)
        scored.append((combined, i, ex))
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return [ex for _, _, ex in scored[:k]]
