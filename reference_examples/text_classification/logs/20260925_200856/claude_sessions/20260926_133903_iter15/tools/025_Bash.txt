"""
Prototype: suffix-focused retrieval for long inputs.
For long inputs, only the last N chars are used for similarity — the specific entity/facts
rather than the shared boilerplate preamble.
"""
import json, re
from collections import defaultdict

_CJK_RE = re.compile(r'[一-鿿぀-ゟ゠-ヿ＀-￯]')
_WRAPPER_RE = re.compile(r'^\s*\[[^\]]*\]|<eoa>\s*$|\[[^\]]*\]\s*$', re.IGNORECASE)

def _has_cjk(text):
    return bool(_CJK_RE.search(text))

def _tokenize_word(text):
    return frozenset(re.findall(r"[A-Za-z0-9]+", text.lower()))

def _tokenize_bigram(text):
    t = re.sub(r'\s+', ' ', text.strip())
    if len(t) < 2:
        return frozenset([t]) if t else frozenset()
    return frozenset(t[i:i+2] for i in range(len(t)-1))

def _tokenize_adaptive(text):
    return _tokenize_bigram(text) if _has_cjk(text) else _tokenize_word(text)

def _suffix_tokens(text, suffix_chars):
    """Use suffix for long inputs, full for short."""
    chunk = text[-suffix_chars:] if len(text) > suffix_chars else text
    return _tokenize_adaptive(chunk)

def _jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

# Build suffix-tokenized examples
def make_pool_suffix(examples_raw, suffix_chars):
    pool = []
    for ex in examples_raw:
        restored = dict(ex)
        raw_q = restored.get('raw_question', restored.get('input', ''))
        restored['tokens_suffix'] = _suffix_tokens(raw_q, suffix_chars)
        pool.append(restored)
    return pool

def get_top_k_suffix(query, pool, suffix_chars, k=8):
    q_tok = _suffix_tokens(query, suffix_chars)
    scored = [(_jaccard(q_tok, ex['tokens_suffix']), i, ex) for i, ex in enumerate(pool)]
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return [ex for _, _, ex in scored[:k]]

def get_top_k_full(query, pool, k=8):
    q_tok = _tokenize_adaptive(query)
    scored = [(_jaccard(q_tok, ex['tokens']), i, ex) for i, ex in enumerate(pool)]
    scored.sort(key=lambda x: (x[0], x[1]), reverse=True)
    return [ex for _, _, ex in scored[:k]]
