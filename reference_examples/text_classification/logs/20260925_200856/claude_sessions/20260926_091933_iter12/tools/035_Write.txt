"""Prototype v3: correct-only pool and recency-stratified interleaving."""
import json, re
from collections import defaultdict
from pathlib import Path

BASE = Path(__file__).parent / "logs" / "20260925_200856"

_CJK_RE = re.compile(r'[一-鿿぀-ゟ゠-ヿ＀-￯]')
def _has_cjk(t): return bool(_CJK_RE.search(t))
def _tok_word(t): return frozenset(re.findall(r"[A-Za-z0-9]+", t.lower()))
def _tok_bigram(t):
    t = re.sub(r'\s+', ' ', t.strip())
    return frozenset(t[i:i+2] for i in range(len(t)-1)) if len(t) >= 2 else frozenset()
def _tok(t): return _tok_bigram(t) if _has_cjk(t) else _tok_word(t)
def _jaccard(a, b):
    if not a and not b: return 0.0
    return len(a & b) / len(a | b)

def load_steps(dataset, system):
    p = BASE / dataset / system / "gpt-oss-120b" / "log.jsonl"
    return [json.loads(l) for l in p.read_text(encoding="utf-8").splitlines()
            if json.loads(l)["type"] == "step"]

steps_law = load_steps("LawBench", "adaptive_tokenizer_confusion_memory")
steps_s2d = load_steps("Symptom2Disease", "adaptive_tokenizer_confusion_memory")

def build_memory(steps, train_n):
    correct, errors, all_ex = [], [], []
    for s in steps[:train_n]:
        raw_q = s["input_preview"][:600]
        tgt = s["tgt"]
        ex = {"tokens": _tok(raw_q), "target": tgt, "ok": s["ok"]}
        all_ex.append(ex)
        (correct if s["ok"] else errors).append(ex)
    return all_ex, correct, errors

all_law, correct_law, errors_law = build_memory(steps_law, 150)
all_s2d, correct_s2d, errors_s2d = build_memory(steps_s2d, 150)

print(f"LawBench: {len(all_law)} total, {len(correct_law)} correct ({100*len(correct_law)//len(all_law)}%), {len(errors_law)} errors")
print(f"Symptom2Disease: {len(all_s2d)} total, {len(correct_s2d)} correct ({100*len(correct_s2d)//len(all_s2d)}%), {len(errors_s2d)} errors")

# ---- Ranking methods ----
def rank_baseline(query_tok, pool):
    return sorted(range(len(pool)), key=lambda i: -_jaccard(query_tok, pool[i]["tokens"]))

def rank_correct_primary(query_tok, correct, errors, correct_frac=0.8):
    """Fill correct_frac of budget from correct pool, rest from errors (similarity-ranked)."""
    n_correct_slots = max(1, int(20 * correct_frac))  # top-20 budget
    c_rank = sorted(range(len(correct)), key=lambda i: -_jaccard(query_tok, correct[i]["tokens"]))
    e_rank = sorted(range(len(errors)),  key=lambda i: -_jaccard(query_tok, errors[i]["tokens"]))
    # Combine: top-n from correct, then top-k from errors, map back to original pool indices
    result = []
    for ci in c_rank[:n_correct_slots]:
        result.append(("c", ci))
    for ei in e_rank[:20 - len(result)]:
        result.append(("e", ei))
    return result

def rank_recency_interleave(query_tok, pool):
    """Split into early (first half) and recent (second half), interleave top-k from each."""
    mid = len(pool) // 2
    early = pool[:mid]
    recent = pool[mid:]
    early_rank  = sorted(range(len(early)),  key=lambda i: -_jaccard(query_tok, early[i]["tokens"]))
    recent_rank = sorted(range(len(recent)), key=lambda i: -_jaccard(query_tok, recent[i]["tokens"]))
    # Interleave: recent-first (alternating recent, early)
    result, ei, ri = [], 0, 0
    while ei < len(early_rank) or ri < len(recent_rank):
        if ri < len(recent_rank):
            result.append(mid + recent_rank[ri]); ri += 1
        if ei < len(early_rank):
            result.append(early_rank[ei]); ei += 1
    return result

# ---- Evaluate ----
def eval_method_pool(test_steps, pool, rank_fn, name):
    hits, ranks = 0, []
    for s in test_steps:
        query_tok = _tok(s["input_preview"][:600])
        tgt = s["tgt"]
        ranking = rank_fn(query_tok)
        for pos, idx in enumerate(ranking[:80]):
            if pool[idx]["target"] == tgt:
                ranks.append(pos + 1); hits += 1; break
    avg = sum(ranks)/len(ranks) if ranks else float('inf')
    mrr = sum(1/r for r in ranks)/len(test_steps) if ranks else 0.0
    print(f"  {name:<38} rank@hit={avg:5.1f}  MRR={mrr:.3f}  found={hits}/{len(test_steps)}")

def eval_correct_primary(test_steps, correct, errors, all_pool, frac, name):
    """Evaluate correct-primary: rank within correct+error pools, match target anywhere."""
    hits, ranks = 0, []
    for s in test_steps:
        query_tok = _tok(s["input_preview"][:600])
        tgt = s["tgt"]
        combined = rank_correct_primary(query_tok, correct, errors, frac)
        for pos, (pool_tag, idx) in enumerate(combined[:80]):
            ex = correct[idx] if pool_tag == "c" else errors[idx]
            if ex["target"] == tgt:
                ranks.append(pos + 1); hits += 1; break
    avg = sum(ranks)/len(ranks) if ranks else float('inf')
    mrr = sum(1/r for r in ranks)/len(test_steps) if ranks else 0.0
    print(f"  {name:<38} rank@hit={avg:5.1f}  MRR={mrr:.3f}  found={hits}/{len(test_steps)}")

print("\n=== LawBench (test 150-200) ===")
test_law = steps_law[150:200]
eval_method_pool(test_law, all_law, lambda q: rank_baseline(q, all_law), "baseline (all examples)")
eval_method_pool(test_law, correct_law, lambda q: rank_baseline(q, correct_law), "baseline (correct-only)")
eval_correct_primary(test_law, correct_law, errors_law, all_law, 0.7, "correct-primary 70/30")
eval_correct_primary(test_law, correct_law, errors_law, all_law, 0.8, "correct-primary 80/20")
eval_correct_primary(test_law, correct_law, errors_law, all_law, 1.0, "correct-only (no errors)")
eval_method_pool(test_law, all_law, lambda q: rank_recency_interleave(q, all_law), "recency interleave")

print("\n=== Symptom2Disease (test 150-200) ===")
test_s2d = steps_s2d[150:200]
eval_method_pool(test_s2d, all_s2d, lambda q: rank_baseline(q, all_s2d), "baseline (all examples)")
eval_method_pool(test_s2d, correct_s2d, lambda q: rank_baseline(q, correct_s2d), "baseline (correct-only)")
eval_correct_primary(test_s2d, correct_s2d, errors_s2d, all_s2d, 0.8, "correct-primary 80/20")
eval_correct_primary(test_s2d, correct_s2d, errors_s2d, all_s2d, 1.0, "correct-only (no errors)")
eval_method_pool(test_s2d, all_s2d, lambda q: rank_recency_interleave(q, all_s2d), "recency interleave")

print("\n=== Label coverage: how many unique targets in correct vs full pool? ===")
for name, pool, full in [("LawBench", correct_law, all_law), ("S2D", correct_s2d, all_s2d)]:
    correct_labels = set(e["target"] for e in pool)
    full_labels = set(e["target"] for e in full)
    print(f"  {name}: correct_pool={len(correct_labels)} labels, full={len(full_labels)}, "
          f"lost_by_correct_filter={len(full_labels - correct_labels)}")

print("\nDone.")
