80	                 adj: dict[str, set[str]],
81	                 max_size: int) -> set[str]:
82	    """BFS from any seed label over the undirected confusion graph.
83	
84	    Returns the union of connected-component members reachable from any seed,
85	    capped at max_size by priority-queue ordering (seeds first, then by
86	    discovery order).
87	    """
88	    visited: set[str] = set()
89	    queue: deque[str] = deque()
90	    for lbl in seed_labels:
91	        if lbl in adj:
92	            visited.add(lbl)
93	            queue.append(lbl)
94	    while queue and len(visited) < max_size:
95	        node = queue.popleft()
96	        for neighbour in adj.get(node, []):
97	            if neighbour not in visited and len(visited) < max_size:
98	                visited.add(neighbour)
99	                queue.append(neighbour)
